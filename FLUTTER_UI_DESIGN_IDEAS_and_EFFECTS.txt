1. GLASSMORPHISM    

Glassmorphism is a modern design (usanifu) style that makes your app look like 
frosted glass (kioo kilichopigwa mvuke) by using semi-transparent backgrounds 
(vifuniko vinavyopitisha mwanga kidogo) and a soft background blur (ukungu mwororo kwa nyuma). 
It utilizes multiple layers (tabaka nyingi) and thin borders (mipaka membamba) to make 
interface elements appear as if they are floating over vibrant colors (rangi zinazong'aa).
 While it is excellent for highlighting headers (vichwa vya habari) or popups (madirisha 
 yanayojitokeza) , it must be used carefully to prevent text from becoming hard to read or 
 causing performance issues (matatizo ya uwezo wa utendaji) on older devices (simu za zamani).

 ****Sample code 1:
 import 'dart:ui';
import 'package:flutter/material.dart';

class GlassCard extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Container(
        // Vibrant background (rangi zinazong'aa)
        decoration: BoxDecoration(
          gradient: LinearGradient(
            colors: [Colors.blue, Colors.purple],
            begin: Alignment.topLeft,
            end: Alignment.bottomRight,
          ),
        ),
        child: Center(
          child: ClipRRect(
            // Ensures the blur stays inside the rounded corners
            borderRadius: BorderRadius.circular(20), [cite: 11]
            child: BackdropFilter(
              // The "secret sauce" for the frosted look (ukungu mwororo)
              filter: ImageFilter.blur(sigmaX: 10, sigmaY: 10), [cite: 9, 12]
              child: Container(
                width: 300,
                height: 200,
                decoration: BoxDecoration(
                  // Semi-transparent background (vifuniko vinavyopitisha mwanga kidogo)
                  color: Colors.white.withOpacity(0.1), [cite: 5, 14]
                  borderRadius: BorderRadius.circular(20),
                  // Thin border (mipaka membamba)
                  border: Border.all(
                    color: Colors.white.withOpacity(0.2), [cite: 8, 13]
                  ),
                ),
                child: Center(
                  child: Text(
                    "Glass Card",
                    style: TextStyle(color: Colors.white, fontSize: 24),
                  ),
                ),
              ),
            ),
          ),
        ),
      ),
    );
  }
}

>>>>>>>>>>>>>class DynamicTabBlobBackground extends StatelessWidget {
  final Color activeColor;
  final bool isDark;
  final int currentIndex;

  const DynamicTabBlobBackground({super.key, 
    required this.activeColor,
    required this.isDark,
    required this.currentIndex,
  });

  @override
  Widget build(BuildContext context) {
    final double primaryOpacity = 0.65;
    final double secondaryOpacity = 0.55;
    final double tertiaryOpacity = 0.45;
    final double extraOpacity = 0.35;
    final double fifthOpacity = 0.25;

    return Container(
      decoration: BoxDecoration(
        gradient: LinearGradient(
          begin: Alignment.topCenter,
          end: Alignment.bottomCenter,
          colors: [
            Color(0xFF022C22),
            Color(0xFF011A13),
          ],
        ),
      ),
      child: Stack(
        children: [
          // Blob 1 - Top Left
          Positioned(
            top: -120,
            left: -100,
            child: _AnimatedBlob(
              color: activeColor.withOpacity(primaryOpacity),
              size: 520,
              offsetX: 80,
              offsetY: 100,
              duration: 7,
            ),
          ),
          // Blob 2 - Bottom Right
          Positioned(
            bottom: -100,
            right: -120,
            child: _AnimatedBlob(
              color: activeColor.withOpacity(secondaryOpacity),
              size: 480,
              offsetX: -90,
              offsetY: -80,
              duration: 9,
            ),
          ),
          // Blob 3 - Center Right
          Positioned(
            top: MediaQuery.of(context).size.height * 0.35,
            right: -140,
            child: _AnimatedBlob(
              color: activeColor.withOpacity(tertiaryOpacity),
              size: 420,
              offsetX: -70,
              offsetY: 60,
              duration: 11,
            ),
          ),
          // Blob 4 - Bottom Left
          Positioned(
            bottom: 100,
            left: -160,
            child: _AnimatedBlob(
              color: activeColor.withOpacity(extraOpacity),
              size: 440,
              offsetX: 100,
              offsetY: -70,
              duration: 13,
            ),
          ),
          // Blob 5 - Top Right
          Positioned(
            top: 30,
            right: -80,
            child: _AnimatedBlob(
              color: activeColor.withOpacity(fifthOpacity),
              size: 360,
              offsetX: -50,
              offsetY: 50,
              duration: 15,
            ),
          ),
          // Blob 6 - Center Left
          Positioned(
            top: MediaQuery.of(context).size.height * 0.6,
            left: -120,
            child: _AnimatedBlob(
              color: activeColor.withOpacity(0.3),
              size: 380,
              offsetX: 70,
              offsetY: -40,
              duration: 17,
            ),
          ),
          // Blur Effect
          BackdropFilter(
            filter: ImageFilter.blur(sigmaX: 50, sigmaY: 50),
            child: const SizedBox.expand(),
          ),
        ],
      ),
    );
  }
